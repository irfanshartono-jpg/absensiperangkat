Panduan Sync Ke Google Sheets (Google Apps Script)

Salin skrip `Code.gs` ini ke Google Sheets Apps Script Anda untuk integrasi cloud gratis.

// ================================================================
// GOOGLE APPS SCRIPT (Code.gs) - PEMDES KEDUNGMAKAM ATTENDANCE API
// ================================================================

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // Header jika sheet masih kosong
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Timestamp", "Tanggal", "Jam", "NIK", "Nama", "Jabatan", "Jarak (m)", "Status", "Device ID", "Catatan"]);
    }
    
    // Append data presensi baru
    sheet.appendRow([
      new Date(),
      data.tanggal,
      data.jam,
      data.nik,
      data.nama,
      data.jabatan,
      data.jarak,
      data.status,
      data.deviceId,
      data.catatan || ''
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({"result": "success", "message": "Absen berhasil disimpan"}))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({"result": "error", "error": error.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

Langkah Penyebaran Web App:

  1. Buka Google Sheets baru → Klik menu Ekstensi → Apps Script.
  2. Hapus isi berkas `Code.gs`, tempelkan kode di atas, lalu simpan (Ctrl+S).
  3. Klik tombol Deploy (Terapkan) → Deployment baru → Jenis: Aplikasi Web.
  4. Akses: Pilih Siapa Saja (Anyone), lalu klik Terapkan dan salin URL Web App Anda.
